## JUDGMENT decisions (locked; rationale for reviewers)

1. **Glass is hand-built over `.ultraThinMaterial`, not iOS 26 `.glassEffect()`.** Deployment target is iOS 17; the §1 recipe (135° white gradient `.66→.4` + `blur(20) saturate(1.7)` + a specific white hairline + two inset highlights) is owner-locked "copy verbatim". `.ultraThinMaterial` supplies the backdrop blur; the white gradient, hairline border, and top inset-highlight are layered on to reproduce the recipe. The native API is system-controlled (can't hit the exact recipe) and would force `#available(iOS 26)`. Same call for every glass component.
2. **Fonts: bundle the variable TTFs; select weights via SwiftUI `Font.custom(family:).weight(_:)` (family + weight-trait resolution).** Google publishes only variable Archivo / Source Serif 4 (no static weights); `fonttools` is unavailable. A CoreText probe confirmed that requesting family `Archivo` / `Source Serif 4` + a weight trait resolves to the correct named instance (400→Regular, 600→SemiBold, 800→ExtraBold) and family + slant → the true italic face — handling the fonts' inconsistent PostScript naming automatically. Registered via `UIAppFonts` in `info.properties`, plus a defensive idempotent `DSFonts.register()` (also lets a host-less unit test load them via a test-target Fonts resource).
3. **Dark palette is a takeover theme via `\.dsPalette`, not iOS dark mode.** The app is light; session screens (F5/F6) opt in with `.dsTheme(.dark)`. The gallery renders both by scoping subtrees.

```swift
/*
 * Purpose: The myCase color token layer — the ONLY place hex literals live.
 *          Two palettes: .light (default, every tab) and .dark (session-takeover
 *          theme, NOT iOS dark mode), threaded through the \.dsPalette environment.
 * Inputs: none.
 * Outputs: DSPalette, Color(hex:), \.dsPalette environment, View.dsTheme(_:).
 * Run: imported by every DesignSystem component; screens read \.dsPalette.
 */

import SwiftUI

struct DSPalette: Equatable {
    let page: Color
    let ink: Color
    let onInk: Color        // chalk text ON an ink fill (toast, CASE circle, avatar)
    let muted: Color        // owner-locked #515A66 in light
    let faint: Color
    let hairline: Color
    let hairlineSoft: Color
    let surface: Color
    let green: Color        // scarce accent
    let link: Color         // cobalt
    let isDark: Bool

    static let light = DSPalette(
        page: Color(hex: 0xEFF2F6),
        ink: Color(hex: 0x0D1C31),
        onInk: Color(hex: 0xF3F5F8),
        muted: Color(hex: 0x515A66),
        faint: Color(hex: 0xA9B4C4),
        hairline: Color(hex: 0xC9D2DF),
        hairlineSoft: Color(hex: 0xDDE3EB),
        surface: Color(hex: 0xFFFFFF),
        green: Color(hex: 0x1B9A5F),
        link: Color(hex: 0x2E56C0),
        isDark: false
    )

    static let dark = DSPalette(
        page: Color(hex: 0x081222),
        ink: Color(hex: 0xE9EEF5),
        onInk: Color(hex: 0x081222),
        muted: Color(hex: 0x7C8CA8),
        faint: Color(hex: 0x3D5075),
        hairline: Color(hex: 0x24365A),
        hairlineSoft: Color(hex: 0x24365A),
        surface: Color(hex: 0x101E36),
        green: Color(hex: 0x2FC07E),
        link: Color(hex: 0x2E56C0),
        isDark: true
    )
}

extension Color {
    /// 0xRRGGBB hex. Confined to the token layer (F0 no-hex-elsewhere rule).
    init(hex: UInt32, alpha: Double = 1) {
        self.init(
            .sRGB,
            red: Double((hex >> 16) & 0xFF) / 255,
            green: Double((hex >> 8) & 0xFF) / 255,
            blue: Double(hex & 0xFF) / 255,
            opacity: alpha
        )
    }
}

private struct DSPaletteKey: EnvironmentKey {
    static let defaultValue = DSPalette.light
}
extension EnvironmentValues {
    var dsPalette: DSPalette {
        get { self[DSPaletteKey.self] }
        set { self[DSPaletteKey.self] = newValue }
    }
}
extension View {
    /// Scope a subtree to a palette (default .light; .dark = session takeover).
    func dsTheme(_ palette: DSPalette) -> some View { environment(\.dsPalette, palette) }
}
```

```swift
/*
 * Purpose: myCase type system — registers the bundled variable fonts and
 *          exposes weight-parameterized helpers plus the §1 type scale.
 * Inputs: bundled Archivo / Source Serif 4 variable TTFs.
 * Outputs: DSFonts.register(), Font.archivo/.serifVoice, DSTextStyle, .dsText.
 * Run: DSFonts.register() at launch; screens call Font.archivo(...)/.serifVoice(...).
 */

import SwiftUI
import UIKit
import CoreText

enum DSFonts {
    /// Idempotent programmatic registration (belt-and-suspenders alongside
    /// UIAppFonts; also loads fonts in the host-less test process).
    static func register() {
        guard !hasRegistered else { return }
        hasRegistered = true
        let names = ["Archivo-Variable", "SourceSerif4-Variable", "SourceSerif4-Italic-Variable"]
        for name in names {
            guard let url = Bundle(for: DSFontsMarker.self).url(forResource: name, withExtension: "ttf")
                    ?? Bundle.main.url(forResource: name, withExtension: "ttf") else { continue }
            CTFontManagerRegisterFontsForURL(url as CFURL, .process, nil)
        }
    }
    private static var hasRegistered = false
}
private final class DSFontsMarker {}

private let wghtAxis = 0x77676874 // 'wght'

private func variableUIFont(psName: String, family: String, size: CGFloat, weight: CGFloat) -> UIFont {
    DSFonts.register()
    let base = UIFont(name: psName, size: size)
        ?? UIFont(name: family, size: size)
        ?? .systemFont(ofSize: size, weight: uiWeight(weight))
    let desc = base.fontDescriptor.addingAttributes([
        UIFontDescriptor.AttributeName(rawValue: "NSCTFontVariationAttribute"): [wghtAxis: weight]
    ])
    return UIFont(descriptor: desc, size: size)
}

private func uiWeight(_ w: CGFloat) -> UIFont.Weight {
    switch w {
    case ..<450: return .regular
    case ..<550: return .medium
    case ..<650: return .semibold
    case ..<750: return .bold
    default: return .heavy
    }
}

extension Font {
    static func archivo(_ size: CGFloat, weight: CGFloat = 400) -> Font {
        Font(variableUIFont(psName: "Archivo-Regular", family: "Archivo", size: size, weight: weight))
    }
    static func serifVoice(_ size: CGFloat, weight: CGFloat = 400, italic: Bool = false) -> Font {
        let ps = italic ? "SourceSerif4-Italic" : "SourceSerif4-Regular"
        return Font(variableUIFont(psName: ps, family: "Source Serif 4", size: size, weight: weight))
    }
}

/// A type-scale entry: font + letter tracking (points) + line spacing.
struct DSTextStyle {
    let font: Font
    let tracking: CGFloat
    let lineSpacing: CGFloat

    // §1 phone type scale. tracking = em × size (SwiftUI tracking is in points).
    static let h1Tab           = DSTextStyle(font: .archivo(28, weight: 800), tracking: -0.84, lineSpacing: 0)   // -.03em
    static let takeoverDisplay = DSTextStyle(font: .archivo(32, weight: 800), tracking: -1.12, lineSpacing: 0)   // -.035em
    static let cardTitle       = DSTextStyle(font: .archivo(20, weight: 700), tracking: -0.2,  lineSpacing: 0)
    static let rowTitle        = DSTextStyle(font: .archivo(14, weight: 600), tracking: 0,     lineSpacing: 0)
    static let rowTitleStrong  = DSTextStyle(font: .archivo(14.5, weight: 700), tracking: 0,   lineSpacing: 0)
    static let kicker          = DSTextStyle(font: .archivo(10, weight: 600), tracking: 1.6,   lineSpacing: 0)  // .16em caps
    static let meta            = DSTextStyle(font: .archivo(11.5, weight: 400), tracking: 0,   lineSpacing: 0)
    static let timerLarge      = DSTextStyle(font: .archivo(34, weight: 700), tracking: 0,     lineSpacing: 0)

    static func serif(_ size: CGFloat, italic: Bool = false, weight: CGFloat = 400) -> DSTextStyle {
        DSTextStyle(font: .serifVoice(size, weight: weight, italic: italic), tracking: 0, lineSpacing: size * 0.5)
    }
}

extension View {
    /// Apply a DSTextStyle (font + tracking + line spacing; color left to caller).
    func dsText(_ s: DSTextStyle) -> some View {
        self.font(s.font).tracking(s.tracking).lineSpacing(s.lineSpacing)
    }
    /// Tabular numerals — always on for timers/scores.
    func tabularNumbers() -> some View { self.monospacedDigit() }
}
```

**Code — `Glass.swift`:** (recipe §1 verbatim; reduce-transparency → solid `body.solid` analog; dark recipe when `palette.isDark`)
```swift
/*
 * Purpose: The glass chrome layer — panels, chips, sheets — reproducing the §1
 *          recipes over .ultraThinMaterial, with a dark-takeover recipe and a
 *          solid-white fallback under Reduce Transparency (the body.solid analog).
 * Inputs: \.dsPalette, \.accessibilityReduceTransparency.
 * Outputs: .glassPanel/.glassChip/.glassSheet, DSScrim, AnyTransition.dsRise.
 * Run: `SomeView().glassPanel()`; sheets `.transition(.dsRise)`.
 */

import SwiftUI

enum GlassKind { case panel, chip, sheet }

struct GlassSurface<S: Shape>: View {
    @Environment(\.dsPalette) private var palette
    @Environment(\.accessibilityReduceTransparency) private var reduceTransparency
    let shape: S
    let kind: GlassKind

    var body: some View {
        ZStack {
            if reduceTransparency {
                shape.fill(palette.surface)                       // --gbg: #FFFFFF; --gblur: 0
            } else if palette.isDark {
                shape.fill(.ultraThinMaterial)
                shape.fill(palette.surface.opacity(0.6))          // rgba(16,30,54,.6)
            } else {
                shape.fill(.ultraThinMaterial)
                shape.fill(gbg)                                   // white gradient .66→.4
            }
        }
        .overlay(shape.strokeBorder(border, lineWidth: 1))
        .overlay(topInsetHighlight)                               // inset 0 1.5px 0 rgba(255,255,255,.85)
        .compositingGroup()
        .shadow(color: shadowColor, radius: shadowRadius, x: 0, y: shadowY)
    }

    private var gbg: LinearGradient {
        LinearGradient(colors: [.white.opacity(0.66), .white.opacity(0.40)],
                       startPoint: .topLeading, endPoint: .bottomTrailing)
    }
    private var border: Color {
        palette.isDark ? palette.ink.opacity(0.14) : .white.opacity(kind == .chip ? 0.75 : 0.72)
    }
    private var topInsetHighlight: some View {
        shape.strokeBorder(
            LinearGradient(colors: [.white.opacity(palette.isDark ? 0.12 : 0.85), .clear],
                           startPoint: .top, endPoint: .center),
            lineWidth: 1.5)
        .blendMode(.plusLighter)
    }
    private var shadowColor: Color {
        palette.isDark ? .black.opacity(0.35) : Color(hex: 0x0D1C31).opacity(kind == .chip ? 0.07 : 0.16)
    }
    private var shadowRadius: CGFloat { kind == .chip ? 8 : (kind == .sheet ? 30 : 22) }
    private var shadowY: CGFloat { kind == .chip ? 2 : (kind == .sheet ? 24 : 18) }
}

extension View {
    func glassPanel(cornerRadius: CGFloat = 28) -> some View {
        background(GlassSurface(shape: RoundedRectangle(cornerRadius: cornerRadius, style: .continuous), kind: .panel))
    }
    func glassChip() -> some View {
        background(GlassSurface(shape: Capsule(), kind: .chip))
    }
    func glassSheet(cornerRadius: CGFloat = 32) -> some View {
        background(GlassSurface(shape: RoundedRectangle(cornerRadius: cornerRadius, style: .continuous), kind: .sheet))
    }
}

/// Scrim under sheets: rgba(13,28,49,.34) + blur(5).
struct DSScrim: View {
    @Environment(\.accessibilityReduceTransparency) private var reduceTransparency
    var body: some View {
        Color(hex: 0x0D1C31).opacity(reduceTransparency ? 0.5 : 0.34)
            .background(reduceTransparency ? nil : .ultraThinMaterial)
            .ignoresSafeArea()
    }
}

extension AnyTransition {
    /// Sheet rise: 16px up + fade (pair with DSMotion.sheetCurve at the call site).
    static var dsRise: AnyTransition {
        .move(edge: .bottom).combined(with: .opacity)
    }
}
```

```swift
/*
 * Purpose: The staircase mark — the only icon in the system. Shape + a gradient,
 *          optionally trim-draw-animated, view.
 * Inputs: \.dsPalette.
 * Outputs: StaircaseMark (Shape), StaircaseMarkView.
 * Run: `StaircaseMarkView(animated: true).frame(width: 24, height: 20)`.
 */

import SwiftUI

struct StaircaseMark: Shape {
    // Points from the SVG path in the native 48×40 viewBox, scaled to `rect`.
    func path(in rect: CGRect) -> Path {
        let sx = rect.width / 48, sy = rect.height / 40
        func p(_ x: CGFloat, _ y: CGFloat) -> CGPoint {
            CGPoint(x: rect.minX + x * sx, y: rect.minY + y * sy)
        }
        var path = Path()
        path.move(to: p(4, 35))
        path.addLine(to: p(15, 35)); path.addLine(to: p(15, 25))
        path.addLine(to: p(26, 25)); path.addLine(to: p(26, 15))
        path.addLine(to: p(37, 15)); path.addLine(to: p(37, 4))
        path.addLine(to: p(45, 4))
        return path
    }
}

struct StaircaseMarkView: View {
    @Environment(\.dsPalette) private var palette
    var lineWidth: CGFloat = 5
    var animated: Bool = false
    /// Optional explicit colors (e.g. CASE circle: chalk→green on ink).
    var lowColor: Color? = nil
    var highColor: Color? = nil
    @State private var trim: CGFloat = 0

    var body: some View {
        StaircaseMark()
            .trim(from: 0, to: animated ? trim : 1)
            .stroke(
                LinearGradient(colors: [lowColor ?? palette.ink, highColor ?? palette.green],
                               startPoint: .bottom, endPoint: .top),
                style: StrokeStyle(lineWidth: lineWidth, lineCap: .butt, lineJoin: .miter))
            .aspectRatio(48.0 / 40.0, contentMode: .fit)
            .onAppear { if animated { withAnimation(DSMotion.drawCurve) { trim = 1 } } }
    }
}
```
